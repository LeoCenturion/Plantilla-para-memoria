      \begin{itemize}

        \item Barrera inferior: límite de pérdida
        \item Barrera temporal: expiración
        \item Adaptativa a la volatilidad dinámica

      \begin{itemize}
        \item K-Fold estándar: \alert{no válido} en series de tiempo
        \item Purgado + embargo (1–5\%)
      \end{itemize}
